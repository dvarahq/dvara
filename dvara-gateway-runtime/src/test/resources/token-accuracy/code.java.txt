public final class RateWindow {
    private final long windowMillis;
    private final int limit;
    private final java.util.ArrayDeque<Long> hits = new java.util.ArrayDeque<>();

    public RateWindow(long windowMillis, int limit) {
        if (windowMillis <= 0 || limit <= 0) {
            throw new IllegalArgumentException("window and limit must be positive");
        }
        this.windowMillis = windowMillis;
        this.limit = limit;
    }

    /** Records a hit at {@code now} and says whether it fits in the window. */
    public synchronized boolean tryAcquire(long now) {
        while (!hits.isEmpty() && now - hits.peekFirst() >= windowMillis) {
            hits.pollFirst();
        }
        if (hits.size() >= limit) {
            return false;
        }
        hits.addLast(now);
        return true;
    }

    @Override
    public String toString() {
        return "RateWindow{windowMillis=" + windowMillis + ", limit=" + limit + ", inWindow=" + hits.size() + "}";
    }
}
